cmake_minimum_required(VERSION 3.20)
project(aikernel_whisper LANGUAGES C CXX)

if(NOT IS_DIRECTORY "${WHISPER_SOURCE_DIR}")
    message(FATAL_ERROR "Set WHISPER_SOURCE_DIR to the pinned whisper.cpp v1.9.5 checkout.")
endif()

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(WHISPER_COREML OFF CACHE BOOL "" FORCE)
set(WHISPER_OPENVINO OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_CUDA OFF CACHE BOOL "" FORCE)
set(GGML_VULKAN OFF CACHE BOOL "" FORCE)
set(GGML_METAL OFF CACHE BOOL "" FORCE)
set(GGML_HIP OFF CACHE BOOL "" FORCE)
set(GGML_SYCL OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_BLAS OFF CACHE BOOL "" FORCE)
set(GGML_LLAMAFILE OFF CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE_DIR}" whisper EXCLUDE_FROM_ALL)

add_library(aikernel_whisper SHARED whisper_bridge.cpp)
target_link_libraries(aikernel_whisper PRIVATE whisper)
target_compile_features(aikernel_whisper PRIVATE cxx_std_17)
target_compile_definitions(aikernel_whisper PRIVATE
    AK_REQUIRES_AVX=$<BOOL:${GGML_AVX}>
    AK_REQUIRES_AVX2=$<BOOL:${GGML_AVX2}>
    AK_REQUIRES_BMI2=$<BOOL:${GGML_BMI2}>
    AK_REQUIRES_FMA=$<BOOL:${GGML_FMA}>
    AK_REQUIRES_F16C=$<BOOL:${GGML_F16C}>)
set_target_properties(aikernel_whisper PROPERTIES CXX_VISIBILITY_PRESET hidden VISIBILITY_INLINES_HIDDEN YES)
install(TARGETS aikernel_whisper RUNTIME DESTINATION . LIBRARY DESTINATION .)
